import sys
import json
from decimal import Decimal
from typing import Dict, Any

from pyspark.context import SparkContext
from pyspark.sql import SparkSession, Row
from awsglue.context import GlueContext


def init_glue_context():
    sc = SparkContext.getOrCreate()
    glueContext = GlueContext(sc)
    spark = glueContext.spark_session
    return glueContext, spark


def parse_dynamodb_json(dynamo_obj):
    """
    Recursively converts DynamoDB JSON to normal Python values.
    Supports S, N, BOOL, NULL, M, L, SS, NS.
    """
    if isinstance(dynamo_obj, dict):
        if "NULL" in dynamo_obj and dynamo_obj["NULL"] is True:
            return None
        if "S" in dynamo_obj:
            return dynamo_obj["S"]
        if "N" in dynamo_obj:
            num_str = dynamo_obj["N"]
            try:
                if "." in num_str:
                    return float(num_str)
                return int(num_str)
            except Exception:
                return num_str
        if "BOOL" in dynamo_obj:
            return dynamo_obj["BOOL"]
        if "SS" in dynamo_obj:
            return dynamo_obj["SS"]
        if "NS" in dynamo_obj:
            out = []
            for n in dynamo_obj["NS"]:
                try:
                    out.append(float(n) if "." in n else int(n))
                except Exception:
                    out.append(n)
            return out
        if "M" in dynamo_obj:
            return {k: parse_dynamodb_json(v) for k, v in dynamo_obj["M"].items()}
        if "L" in dynamo_obj:
            return [parse_dynamodb_json(i) for i in dynamo_obj["L"]]

        # normal dict
        return {k: parse_dynamodb_json(v) for k, v in dynamo_obj.items()}

    elif isinstance(dynamo_obj, list):
        return [parse_dynamodb_json(i) for i in dynamo_obj]

    return dynamo_obj


def flatten_dict(
    d: Dict[str, Any],
    parent: str = "",
    sep: str = "_",
    list_to_json: bool = True,
    index_before_parent: bool = False,
    out: Dict[str, Any] = None
) -> Dict[str, Any]:
    """
    Flatten nested dict to single-level dict.
    Example:
      {"a": {"b": 1}} -> {"a_b": 1}
    """
    if out is None:
        out = {}

    for k, v in d.items():
        new_key = f"{parent}{sep}{k}" if parent else k

        if isinstance(v, dict):
            flatten_dict(
                v,
                parent=new_key,
                sep=sep,
                list_to_json=list_to_json,
                index_before_parent=index_before_parent,
                out=out
            )

        elif isinstance(v, list):
            if all(isinstance(el, dict) for el in v):
                for idx, el in enumerate(v):
                    idx_parent = f"{idx}{sep}{new_key}" if index_before_parent else f"{new_key}{sep}{idx}"
                    flatten_dict(
                        el,
                        parent=idx_parent,
                        sep=sep,
                        list_to_json=list_to_json,
                        index_before_parent=index_before_parent,
                        out=out
                    )
            else:
                out[new_key] = json.dumps(v, ensure_ascii=False) if list_to_json else v

        else:
            out[new_key] = v

    return out


def process_call_entities(glueContext, spark, database: str, table: str, output_path: str, mode: str = "overwrite"):
    # 1) Read from Glue Catalog
    dyf_raw = glueContext.create_dynamic_frame.from_catalog(
        database=database,
        table_name=table
    )

    # 2) Convert DynamicFrame -> Spark DataFrame
    df_raw = dyf_raw.toDF()

    # Debug
    print("Raw schema:")
    df_raw.printSchema()

    # 3) If empty input, fail fast
    if len(df_raw.columns) == 0:
        raise ValueError("Input DataFrame has 0 columns. Check crawler/table schema.")

    if df_raw.rdd.isEmpty():
        raise ValueError("Input DataFrame has no rows. Nothing to transform.")

    # 4) Convert each row -> dict -> parse DynamoDB JSON -> flatten
    parsed_rdd = df_raw.rdd.map(lambda row: row.asDict(recursive=True)) \
                           .map(parse_dynamodb_json) \
                           .map(flatten_dict)

    # 5) If parsing produced nothing useful, stop early
    sample = parsed_rdd.take(1)
    if not sample:
        raise ValueError("No parsed rows produced.")
    if not sample[0]:
        raise ValueError("Parsed rows are empty dicts. Check source DynamoDB JSON structure.")

    # 6) Rebuild Spark DataFrame
    df_flat = spark.createDataFrame(parsed_rdd.map(lambda x: Row(**x)))

    print("Flattened schema:")
    df_flat.printSchema()

    if len(df_flat.columns) == 0:
        raise ValueError("Flattened DataFrame has 0 columns. Cannot write parquet.")

    # 7) Write parquet
    df_flat.write.mode(mode).parquet(output_path)

    print(f"Wrote parquet to {output_path} (mode={mode})")
    return df_flat


# Example usage
glueContext, spark = init_glue_context()

database = "aaa_dashboard_call-entities-raw_database"
table = "call_entities_raw"
output_path = "sbuket/"

df_final = process_call_entities(glueContext, spark, database, table, output_path)
