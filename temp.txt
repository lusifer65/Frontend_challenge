import sys
from awsglue.utils import getResolvedOptions
from awsglue.context import GlueContext
from awsglue.job import Job
from pyspark.context import SparkContext
from pyspark.sql import functions as F
from pyspark.sql.types import (
    StructType, StructField, StringType, BooleanType,
    DoubleType, MapType
)

# -----------------------------------------------------------------------------------
# Glue job setup
# -----------------------------------------------------------------------------------
args = getResolvedOptions(sys.argv, ["JOB_NAME"])

sc = SparkContext()
glueContext = GlueContext(sc)
spark = glueContext.spark_session
job = Job(glueContext)
job.init(args["JOB_NAME"], args)

INPUT_PATH = "s3://dashboard-bucket-aaa/exports/raw_csv/"
OUTPUT_PATH = "s3://dashboard-bucket-aaa/exports/connect_parquet/"

# -----------------------------------------------------------------------------------
# Read raw CSV
# -----------------------------------------------------------------------------------
df = (
    spark.read
    .option("header", "true")
    .option("multiLine", "false")
    .option("escape", '"')
    .csv(INPUT_PATH)
)

# -----------------------------------------------------------------------------------
# Normalize plain "null" strings to actual nulls
# -----------------------------------------------------------------------------------
for c in df.columns:
    df = df.withColumn(
        c,
        F.when(
            (F.trim(F.col(c)) == "null") | (F.trim(F.col(c)) == ""),
            F.lit(None)
        ).otherwise(F.col(c))
    )

# -----------------------------------------------------------------------------------
# JSON Schemas for DynamoDB JSON string columns
# -----------------------------------------------------------------------------------
agent_schema = StructType([
    StructField("Username", StructType([
        StructField("S", StringType(), True)
    ]), True),
    StructField("Id", StructType([
        StructField("S", StringType(), True)
    ]), True)
])

attributes_schema = MapType(
    StringType(),
    StructType([
        StructField("S", StringType(), True),
        StructField("N", StringType(), True),
        StructField("BOOL", BooleanType(), True)
    ]),
    True
)

contact_lens_schema = StructType([
    StructField("Sentiment", StructType([
        StructField("M", StructType([
            StructField("Overall", StructType([
                StructField("S", StringType(), True)
            ]), True),
            StructField("Score", StructType([
                StructField("N", StringType(), True)
            ]), True)
        ]), True)
    ]), True)
])

endpoint_schema = StructType([
    StructField("Address", StructType([
        StructField("S", StringType(), True)
    ]), True),
    StructField("Type", StructType([
        StructField("S", StringType(), True)
    ]), True)
])

metrics_schema = StructType([
    StructField("HoldTime", StructType([
        StructField("N", StringType(), True)
    ]), True),
    StructField("InteractionTime", StructType([
        StructField("N", StringType(), True)
    ]), True),
    StructField("HandleTime", StructType([
        StructField("N", StringType(), True)
    ]), True),
    StructField("AfterContactWorkTime", StructType([
        StructField("N", StringType(), True)
    ]), True),
    StructField("Queue", StructType([
        StructField("M", StructType([
            StructField("Duration", StructType([
                StructField("N", StringType(), True)
            ]), True),
            StructField("EnqueueTimestamp", StructType([
                StructField("S", StringType(), True)
            ]), True)
        ]), True)
    ]), True)
])

queue_schema = StructType([
    StructField("Id", StructType([
        StructField("S", StringType(), True)
    ]), True),
    StructField("Name", StructType([
        StructField("S", StringType(), True)
    ]), True)
])

recording_schema = StructType([
    StructField("Status", StructType([
        StructField("S", StringType(), True)
    ]), True),
    StructField("Location", StructType([
        StructField("S", StringType(), True)
    ]), True)
])

system_endpoint_schema = StructType([
    StructField("Address", StructType([
        StructField("S", StringType(), True)
    ]), True),
    StructField("Type", StructType([
        StructField("S", StringType(), True)
    ]), True)
])

# -----------------------------------------------------------------------------------
# Parse JSON string columns
# -----------------------------------------------------------------------------------
df = (
    df
    .withColumn("Agent_json", F.from_json(F.col("Agent"), agent_schema))
    .withColumn("Attributes_json", F.from_json(F.col("Attributes"), attributes_schema))
    .withColumn("ContactLens_json", F.from_json(F.col("ContactLens"), contact_lens_schema))
    .withColumn("CustomerEndpoint_json", F.from_json(F.col("CustomerEndpoint"), endpoint_schema))
    .withColumn("Metrics_json", F.from_json(F.col("Metrics"), metrics_schema))
    .withColumn("Queue_json", F.from_json(F.col("Queue"), queue_schema))
    .withColumn("Recording_json", F.from_json(F.col("Recording"), recording_schema))
    .withColumn("SystemEndpoint_json", F.from_json(F.col("SystemEndpoint"), system_endpoint_schema))
)

# -----------------------------------------------------------------------------------
# Flatten + type cast
# -----------------------------------------------------------------------------------
df_flat = df.select(
    F.col("pk").alias("pk"),
    F.col("sk").alias("sk"),
    F.col("Channel").alias("channel"),
    F.col("ConnectedToAgentTimestamp").alias("connected_to_agent_timestamp"),
    F.col("ConnectedToSystemTimestamp").alias("connected_to_system_timestamp"),
    F.col("ContactId").alias("contact_id"),
    F.col("DisconnectReason").alias("disconnect_reason"),
    F.col("DisconnectTimestamp").alias("disconnect_timestamp"),
    F.col("InitiationMethod").alias("initiation_method"),
    F.col("InitiationTimestamp").alias("initiation_timestamp"),
    F.col("InstanceId").alias("instance_id"),
    F.col("NextContactId").alias("next_contact_id"),
    F.col("PreviousContactId").alias("previous_contact_id"),

    F.col("Agent_json.Username.S").alias("agent_username"),
    F.col("Agent_json.Id.S").alias("agent_id"),

    F.col("Attributes_json")["issueType"]["S"].alias("issue_type"),
    F.col("Attributes_json")["customerType"]["S"].alias("customer_type"),
    F.col("Attributes_json")["language"]["S"].alias("language"),
    F.col("Attributes_json")["region"]["S"].alias("region"),

    F.col("ContactLens_json.Sentiment.M.Overall.S").alias("sentiment_overall"),
    F.col("ContactLens_json.Sentiment.M.Score.N").cast(DoubleType()).alias("sentiment_score"),

    F.col("CustomerEndpoint_json.Address.S").alias("customer_address"),
    F.col("CustomerEndpoint_json.Type.S").alias("customer_endpoint_type"),

    F.col("Metrics_json.HoldTime.N").cast(DoubleType()).alias("hold_time"),
    F.col("Metrics_json.InteractionTime.N").cast(DoubleType()).alias("interaction_time"),
    F.col("Metrics_json.HandleTime.N").cast(DoubleType()).alias("handle_time"),
    F.col("Metrics_json.AfterContactWorkTime.N").cast(DoubleType()).alias("after_contact_work_time"),
    F.col("Metrics_json.Queue.M.Duration.N").cast(DoubleType()).alias("queue_duration"),
    F.col("Metrics_json.Queue.M.EnqueueTimestamp.S").alias("queue_enqueue_timestamp"),

    F.col("Queue_json.Id.S").alias("queue_id"),
    F.col("Queue_json.Name.S").alias("queue_name"),

    F.col("Recording_json.Status.S").alias("recording_status"),
    F.col("Recording_json.Location.S").alias("recording_location"),

    F.col("SystemEndpoint_json.Address.S").alias("system_address"),
    F.col("SystemEndpoint_json.Type.S").alias("system_type")
)

# -----------------------------------------------------------------------------------
# Add partition column
# -----------------------------------------------------------------------------------
df_flat = (
    df_flat
    .withColumn("event_timestamp", F.to_timestamp("initiation_timestamp", "yyyy-MM-dd'T'HH:mm:ssX"))
    .withColumn("event_date", F.to_date("event_timestamp"))
)

# -----------------------------------------------------------------------------------
# Optional dedupe by contact_id + sk
# -----------------------------------------------------------------------------------
df_flat = df_flat.dropDuplicates(["contact_id", "sk"])

# -----------------------------------------------------------------------------------
# Write parquet partitioned by event_date
# -----------------------------------------------------------------------------------
(
    df_flat.write
    .mode("overwrite")
    .partitionBy("event_date")
    .parquet(OUTPUT_PATH)
)

job.commit()
